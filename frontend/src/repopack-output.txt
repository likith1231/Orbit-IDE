import React, { useState } from 'react';
import { useNavigate, Link } from 'react-router-dom';
import { saveAuth } from '../auth';
import './auth-pages.css';

export default function Login() {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');
  const [loading, setLoading] = useState(false);
  const navigate = useNavigate();

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError('');
    setLoading(true);
    try {
      const res = await fetch('http://localhost:5000/api/auth/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, password }),
      });
      const data = await res.json();
      if (!res.ok) {
        setError(data.error || 'Login failed.');
        return;
      }
      saveAuth(data.token, data.user);
      navigate('/ide');
    } catch {
      setError('Could not reach the server.');
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="auth-page">
      <div className="auth-card">
        <div className="auth-logo">⚡ AI Cloud IDE</div>
        <h1>Welcome back</h1>
        <p className="auth-subtitle">Sign in to continue to your workspace</p>

        <form onSubmit={handleSubmit}>
          <label>Email</label>
          <input type="email" value={email} onChange={(e) => setEmail(e.target.value)} required placeholder="you@example.com" />

          <label>Password</label>
          <input type="password" value={password} onChange={(e) => setPassword(e.target.value)} required placeholder="••••••••" />

          {error && <div className="auth-error">{error}</div>}

          <button type="submit" disabled={loading}>{loading ? 'Signing in…' : 'Sign in'}</button>
        </form>

        <p className="auth-footer">
          Don't have an account? <Link to="/signup">Sign up</Link>
        </p>
      </div>
    </div>
  );
}

const TOKEN_KEY = 'aicloudide_token';
const USER_KEY = 'aicloudide_user';

export function saveAuth(token, user) {
  localStorage.setItem(TOKEN_KEY, token);
  localStorage.setItem(USER_KEY, JSON.stringify(user));
}

export function getToken() {
  return localStorage.getItem(TOKEN_KEY);
}

export function getUser() {
  const raw = localStorage.getItem(USER_KEY);
  return raw ? JSON.parse(raw) : null;
}

export function clearAuth() {
  localStorage.removeItem(TOKEN_KEY);
  localStorage.removeItem(USER_KEY);
}

export function authHeader() {
  const token = getToken();
  return token ? { Authorization: `Bearer ${token}` } : {};
}

      theme: { background: '#1e1e1e', foreground: '#cccccc', cursor: '#cccccc', selectionBackground: '#264f78', black: '#000000', red: '#cd3131', green: '#0dbc79', yellow: '#e5e510', blue: '#2472c8', magenta: '#bc3fbc', cyan: '#11a8cd', white: '#e5e5e5' },
    });
    const fitAddon = new FitAddon();
    term.loadAddon(fitAddon);
    term.open(terminalRef.current);
    fitAddon.fit();
    xtermRef.current = term;
    const obs = new ResizeObserver(() => fitAddon.fit());
    obs.observe(terminalRef.current);
    term.onData((d) => socketRef.current.emit('terminal-input', d));
    socketRef.current.on('terminal-output', (d) => term.write(d));
    socketRef.current.on('connect', () => setDockerStatus('checking'));
    return () => { obs.disconnect(); socketRef.current.disconnect(); term.dispose(); };
  }, []);

  // Docker poll
  useEffect(() => {
    const poll = async () => {
      try {
        const res = await fetch('http://localhost:5000/api/containers/list');
        const data = await res.json();
        setContainers(data.containers || []);
        setDockerStatus(data.containers?.filter(c => c.State === 'running').length ? 'running' : 'idle');
      } catch { setDockerStatus('error'); }
    };
    poll();
    const iv = setInterval(poll, 5000);
    return () => clearInterval(iv);
  }, []);

  // Git poll
  useEffect(() => {
    const fetchGit = async () => {
      try {
        const [sRes, dRes] = await Promise.all([
          fetch('http://localhost:5000/api/git/status'),
          fetch('http://localhost:5000/api/git/diff'),
        ]);
        const s = await sRes.json();
        const d = await dRes.json();
        setGitStatus(s.status || '');
        setGitBranch(s.branch || 'main');
        setGitDiff(d.diff || '');
      } catch {}
    };
    fetchGit();
    const iv = setInterval(fetchGit, 10000);
    return () => clearInterval(iv);
  }, []);

  // Keyboard shortcuts
  useEffect(() => {
    const handler = (e) => {
      const mod = navigator.platform.toUpperCase().includes('MAC') ? e.metaKey : e.ctrlKey;
      if (mod && e.shiftKey && e.key.toLowerCase() === 'p') {
        e.preventDefault(); setPaletteOpen((v) => !v); setPaletteQuery(''); setPaletteIndex(0);
      }
      if (e.key === 'Escape') setPaletteOpen(false);
      if (mod && e.key === '`') { e.preventDefault(); setBottomTab('terminal'); }
    };
    window.addEventListener('keydown', handler);
    return () => window.removeEventListener('keydown', handler);
  }, []);

  useEffect(() => {
    if (chatScrollRef.current) chatScrollRef.current.scrollTop = chatScrollRef.current.scrollHeight;
  }, [chatMessages]);

  // Search
  useEffect(() => {
    if (!searchQuery.trim()) { setSearchResults([]); return; }
    const results = [];
    Object.entries(files).forEach(([name, f]) => {
      f.value?.split('\n').forEach((line, i) => {
        if (line.toLowerCase().includes(searchQuery.toLowerCase()))
          results.push({ file: name, line: i + 1, text: line.trim() });
      });
    });
    setSearchResults(results.slice(0, 80));
  }, [searchQuery, files]);

  const openFile = (name) => {
    setActiveFile(name);
    if (!openTabs.includes(name)) setOpenTabs((t) => [...t, name]);
  };

  const closeTab = (name, e) => {
    e.stopPropagation();
    const next = openTabs.filter((t) => t !== name);
    setOpenTabs(next);
    if (activeFile === name) setActiveFile(next[next.length - 1] || null);
  };

  const saveFile = useCallback(async (fileName, content) => {
    const fileId = files[fileName]?.id;
    if (!project || !fileId) return;
    setSaveStatus('saving');
    try {
      await fetch(`http://localhost:5000/api/projects/${project.id}/files/${fileId}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json', ...authHeader() },
        body: JSON.stringify({ content }),
      });
      setSaveStatus('saved');
    } catch { setSaveStatus('unsaved'); }
  }, [project, files]);

  const updateActiveFileContent = (value) => {
    setFiles((f) => ({ ...f, [activeFile]: { ...f[activeFile], value } }));
    setSaveStatus('unsaved');
    clearTimeout(saveTimeout.current);
    saveTimeout.current = setTimeout(() => saveFile(activeFile, value), 800);
  };

  const applyCodeToFile = useCallback(async (code, fileName) => {
    if (!code || !fileName) return;
    setFiles((f) => ({ ...f, [fileName]: { ...f[fileName], value: code } }));
    await saveFile(fileName, code);
  }, [saveFile]);

  const runFile = useCallback(async (fileName) => {
    const file = fileName || activeFile;
    if (!file) return;
    const code = files[file]?.value || '';
    const language = files[file]?.language;
    setIsRunning(true);
    setBottomTab('terminal');
    xtermRef.current.writeln(`\r\n\x1b[36m▶ Running ${file}...\x1b[0m`);
    try {
      const res = await fetch('http://localhost:5000/api/run', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ code, language, fileName: file }),
      });
      const data = await res.json();
      const output = data.output || data.error || '(no output)';
      output.split('\n').forEach((line) => xtermRef.current.writeln(line));
      setRunOutput(output);
    } catch {
      xtermRef.current.writeln('\r\n\x1b[31m✖ Sandbox execution failed.\x1b[0m');
    } finally { setIsRunning(false); }
  }, [activeFile, files]);

  // AI chat — fully automatic, no permission dialogs
  const sendChatMessage = async () => {
    if (!chatInput.trim() || chatLoading) return;
    const userText = chatInput;
    const nextMessages = [...chatMessages, { role: 'user', content: userText }];
    setChatMessages(nextMessages);
    setChatInput('');
    setChatLoading(true);

    const fileMap = Object.fromEntries(Object.entries(files).map(([n, f]) => [n, f.value]));

    try {
      const res = await fetch('http://localhost:5000/api/ai/chat', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ messages: nextMessages, files: fileMap, activeFile }),
      });
      const data = await res.json();

      if (data.action === 'apply' && data.code && data.fileName) {
        setChatMessages((m) => [...m, { role: 'assistant', content: (data.reply || 'Applying changes...') + `\n\n✔ Applied to ${data.fileName}.` }]);
        await applyCodeToFile(data.code, data.fileName);
      } else if (data.action === 'run' && data.fileName) {
        setChatMessages((m) => [...m, { role: 'assistant', content: `Running ${data.fileName} — check the Terminal tab.` }]);
        await runFile(data.fileName);
      } else {
        setChatMessages((m) => [...m, { role: 'assistant', content: data.reply || data.error || 'No response.' }]);
      }
    } catch {
      setChatMessages((m) => [...m, { role: 'assistant', content: '⚠ Could not reach the AI backend.' }]);
    } finally { setChatLoading(false); }
  };

  const createNewFile = async () => {
    const name = newFileName.trim();
    if (!name || !project) return;
    try {
      const res = await fetch(`http://localhost:5000/api/projects/${project.id}/files`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', ...authHeader() },
        body: JSON.stringify({ name, language: getLang(name) }),
      });
      const data = await res.json();
      setFiles((f) => ({ ...f, [name]: { id: data.file.id, language: getLang(name), value: '' } }));
      setNewFileName(''); setShowNewFile(false);
      openFile(name);
    } catch {}
  };

  const deleteFile = async (name, e) => {
    e.stopPropagation();
    if (!window.confirm(`Delete ${name}?`)) return;
    const fileId = files[name]?.id;
    if (!fileId) return;
    try {
      await fetch(`http://localhost:5000/api/projects/${project.id}/files/${fileId}`, {
        method: 'DELETE', headers: authHeader(),
      });
      setFiles((f) => { const n = { ...f }; delete n[name]; return n; });
      closeTab(name, { stopPropagation: () => {} });
    } catch {}
  };

  const handleLogout = () => { clearAuth(); navigate('/login'); };
  const activeLang = useMemo(() => files[activeFile]?.language || 'plaintext', [files, activeFile]);

  const commands = useMemo(() => [
    { id: 'run', label: '▶ Run Active File', action: () => runFile(activeFile) },
    { id: 'new-file', label: '+ New File', action: () => { setActivityPanel('explorer'); setShowNewFile(true); } },
    { id: 'explorer', label: 'Explorer', action: () => setActivityPanel('explorer') },
    { id: 'search-panel', label: 'Search across files', action: () => setActivityPanel('search') },
    { id: 'docker-panel', label: 'Docker containers', action: () => setActivityPanel('docker') },
    { id: 'ai-panel', label: 'AI Assistant', action: () => setActivityPanel('ai') },
    { id: 'git-panel', label: 'Git status', action: () => setActivityPanel('git') },
    { id: 'terminal-tab', label: 'Terminal', action: () => setBottomTab('terminal') },
    { id: 'problems-tab', label: 'Problems', action: () => setBottomTab('problems') },
    ...Object.keys(files).map((name) => ({ id: `open-${name}`, label: `Open ${name}`, action: () => openFile(name) })),
    { id: 'logout', label: 'Log Out', action: handleLogout },
  ], [files, activeFile, runFile]);

  const fuse = useMemo(() => new Fuse(commands, { keys: ['label'], threshold: 0.4 }), [commands]);
  const filteredCommands = paletteQuery ? fuse.search(paletteQuery).map((r) => r.item) : commands;
  const runCommand = (cmd) => { cmd.action(); setPaletteOpen(false); };
  const handlePaletteKeyDown = (e) => {
    if (e.key === 'ArrowDown') { e.preventDefault(); setPaletteIndex((i) => Math.min(i + 1, filteredCommands.length - 1)); }
    else if (e.key === 'ArrowUp') { e.preventDefault(); setPaletteIndex((i) => Math.max(i - 1, 0)); }
    else if (e.key === 'Enter') { e.preventDefault(); if (filteredCommands[paletteIndex]) runCommand(filteredCommands[paletteIndex]); }
  };

  const gitLines = gitStatus.split('\n').filter(Boolean);
  const errorCount = problems.filter(p => p.severity === 'error').length;
  const warnCount = problems.filter(p => p.severity === 'warning').length;

  return (
    <div className="ide-root">
      <header className="titlebar">
        <div className="titlebar-left">
          <span className="logo">⚡ AI Cloud IDE</span>
          <span className="titlebar-sep">/</span>
          <span className="titlebar-project">{project?.name || '...'}</span>
        </div>
        <div className="titlebar-right">
          <button className="palette-hint-btn" onClick={() => setPaletteOpen(true)}>
            <Command size={13} />&nbsp;Commands
          </button>
          <span className="save-indicator">
            {saveStatus === 'saving' ? '⏳ Saving…' : saveStatus === 'unsaved' ? '● Unsaved' : '✓ Saved'}
          </span>
          <button className="run-btn" onClick={() => runFile(activeFile)} disabled={isRunning}>
            <Play size={12} /> {isRunning ? 'Running…' : 'Run'}
          </button>
          <button className="logout-btn" onClick={handleLogout} title="Log out"><LogOut size={16} /></button>
        </div>
      </header>

      <div className="ide-body">
        <nav className="activity-bar">
          {[
            { id: 'explorer', icon: <Files size={20} />, tip: 'Explorer' },
            { id: 'search', icon: <Search size={20} />, tip: 'Search' },
            { id: 'git', icon: <GitBranch size={20} />, tip: 'Git' },
            { id: 'docker', icon: <Boxes size={20} />, tip: 'Docker Sandbox' },
            { id: 'ai', icon: <Bot size={20} />, tip: 'AI Assistant' },
          ].map((item) => (
            <button key={item.id} className={`activity-icon ${activityPanel === item.id ? 'active' : ''}`}
              onClick={() => setActivityPanel(item.id)} title={item.tip}>
              {item.icon}
            </button>
          ))}
        </nav>

        <PanelGroup direction="horizontal" className="main-panels" style={{ height: '100%', width: '100%' }}>
          <Panel defaultSize={16} minSize={10} maxSize={28} className="sidebar">

            {/* EXPLORER */}
            {activityPanel === 'explorer' && (
              <>
                <div className="sidebar-header">
                  <span className="sidebar-title-text">EXPLORER</span>
                  <button className="icon-btn" onClick={() => setShowNewFile(v => !v)} title="New File"><Plus size={14} /></button>
                </div>
                {showNewFile && (
                  <div className="new-file-row">
                    <input className="new-file-input" autoFocus value={newFileName}
                      onChange={(e) => setNewFileName(e.target.value)}
                      onKeyDown={(e) => { if (e.key === 'Enter') createNewFile(); if (e.key === 'Escape') setShowNewFile(false); }}
                      placeholder="filename.js" />
                    <button className="icon-btn" onClick={createNewFile}>✓</button>
                  </div>
                )}
                <div className="file-tree">
                  <div className="folder-row">▾ {project?.name || 'project'}</div>
                  {Object.keys(files).map((name) => (
                    <div key={name} className={`file-row ${activeFile === name ? 'active' : ''}`} onClick={() => openFile(name)}>
                      <FileDot name={name} />
                      <span className="file-name">{name}</span>
                      <button className="file-delete-btn" onClick={(e) => deleteFile(name, e)} title="Delete"><Trash2 size={11} /></button>
                    </div>
                  ))}
                </div>
                <div className="sidebar-branch">⎇ {gitBranch}</div>
              </>
            )}

            {/* SEARCH */}
            {activityPanel === 'search' && (
              <>
                <div className="sidebar-header"><span className="sidebar-title-text">SEARCH</span></div>
                <div style={{ padding: '8px 12px' }}>
                  <input className="search-input" placeholder="Search across all files..."
                    value={searchQuery} onChange={(e) => setSearchQuery(e.target.value)} autoFocus />
                </div>
                <div className="search-results">
                  {searchResults.map((r, i) => (
                    <div key={i} className="search-result-row" onClick={() => openFile(r.file)}>
                      <div className="search-result-file">{r.file}:{r.line}</div>
                      <div className="search-result-text">{r.text}</div>
                    </div>
                  ))}
                  {searchQuery && !searchResults.length && <div className="no-results">No results found.</div>}
                </div>
              </>
            )}

            {/* GIT */}
            {activityPanel === 'git' && (
              <>
                <div className="sidebar-header">
                  <span className="sidebar-title-text">GIT — {gitBranch}</span>
                  <button className="icon-btn" onClick={() => fetch('http://localhost:5000/api/git/status').then(r=>r.json()).then(d=>{setGitStatus(d.status);setGitBranch(d.branch);})}><RefreshCw size={13}/></button>
                </div>
                <div style={{ padding: '8px 12px', overflowY: 'auto', flex: 1 }}>
                  {!gitLines.length && <div className="no-results">Working tree clean.</div>}
                  {gitLines.map((line, i) => (
                    <div key={i} className="git-status-row">
                      <span className={`git-badge ${line[0]==='M'?'modified':line[0]==='?'?'untracked':'added'}`}>{line[0]==='?'?'U':line[0]}</span>
                      <span style={{fontSize:12}}>{line.slice(3)}</span>
                    </div>
                  ))}
                  {gitDiff && <>
                    <div style={{fontSize:11,color:'var(--vscode-text-dim)',margin:'12px 0 4px'}}>DIFF PREVIEW</div>
                    <pre className="git-diff-view">{gitDiff.slice(0,1500)}{gitDiff.length>1500?'\n...(truncated)':''}</pre>
                  </>}
                </div>
              </>
            )}

            {/* DOCKER */}
            {activityPanel === 'docker' && (
              <>
                <div className="sidebar-header"><span className="sidebar-title-text">DOCKER SANDBOX</span></div>
                <div style={{ padding: '12px 16px' }}>
                  <div style={{fontSize:12,color:'var(--vscode-text-dim)',marginBottom:12,lineHeight:1.5}}>
                    Sandbox containers are spun up automatically when you click ▶ Run. Each run gets a fresh, isolated, network-disabled container with resource limits.
                  </div>
                  <div className={`docker-status ${dockerStatus}`} style={{marginBottom:10}}>● {dockerStatus} ({containers.length} containers)</div>
                  <div className="container-list">
                    {containers.map((c) => (
                      <div key={c.Id} className="container-row">
                        <span className={`container-dot ${c.State}`} />
                        <span style={{fontSize:11}}>{c.Names?.[0]?.replace('/','') || c.Id.slice(0,10)} — {c.State}</span>
                      </div>
                    ))}
                    {!containers.length && <div className="no-results">No containers running.</div>}
                  </div>
                </div>
              </>
            )}

            {/* AI ASSISTANT */}
            {activityPanel === 'ai' && (
              <div className="ai-chat-panel">
                <div className="sidebar-header">
                  <span className="sidebar-title-text">AI ASSISTANT</span>
                  <span className="ai-badge">auto</span>
                </div>
                <div className="chat-messages" ref={chatScrollRef}>
                  {chatMessages.map((m, i) => (
                    <div key={i} className={`chat-msg ${m.role}`}>{m.content}</div>
                  ))}
                  {chatLoading && (
                    <div className="chat-msg assistant">
                      <span className="typing-dot"/><span className="typing-dot"/><span className="typing-dot"/>
                    </div>
                  )}
                </div>
                <div className="chat-input-row">
                  <input className="chat-input" value={chatInput}
                    onChange={(e) => setChatInput(e.target.value)}
                    onKeyDown={(e) => e.key === 'Enter' && !e.shiftKey && sendChatMessage()}
                    placeholder="Ask or command the AI..." />
                  <button className="chat-send-btn" onClick={sendChatMessage} disabled={chatLoading}><Send size={14}/></button>
                </div>
              </div>
            )}
          </Panel>
          <PanelResizeHandle className="resize-handle" />

          {/* CENTER: EDITOR + BOTTOM PANEL */}
          <Panel defaultSize={70} minSize={40}>
            <PanelGroup direction="vertical">
              <Panel defaultSize={70} minSize={20}>
                <div className="tabs-bar">
                  {openTabs.map((name) => (
                    <div key={name} className={`tab ${activeFile === name ? 'active' : ''}`} onClick={() => setActiveFile(name)}>
                      <FileDot name={name}/> {name}
                      <span className="tab-close" onClick={(e) => closeTab(name, e)}>✕</span>
                    </div>
                  ))}
                </div>
                <div className="breadcrumb">
                  <span>{project?.name}</span>
                  <ChevronRight size={12}/>
                  <span>{activeFile}</span>
                </div>
                {activeFile ? (
                  <Editor
                    height="calc(100% - 59px)"
                    language={activeLang}
                    theme="vs-dark"
                    path={activeFile}
                    value={files[activeFile]?.value || ''}
                    onChange={updateActiveFileContent}
                    onMount={(editor, monaco) => {
                      editor.onDidChangeModelDecorations(() => {
                        const model = editor.getModel();
                        if (!model) return;
                        const markers = monaco.editor.getModelMarkers({ resource: model.uri });
                        setProblems(markers.map((m) => ({
                          file: activeFile, line: m.startLineNumber, col: m.startColumn,
                          message: m.message, severity: m.severity === 8 ? 'error' : 'warning',
                        })));
                      });
                    }}
                    options={{
                      fontSize: 14, fontFamily: "'JetBrains Mono', Consolas, monospace",
                      fontLigatures: true, minimap: { enabled: true },
                      smoothScrolling: true, cursorBlinking: 'smooth',
                      padding: { top: 12 }, wordWrap: 'on',
                      scrollBeyondLastLine: false,
                      suggest: { showKeywords: true, showSnippets: true },
                    }}
                  />
                ) : (
                  <div className="editor-empty">
                    <div style={{fontSize:32}}>⚡</div>
                    <div style={{fontSize:18,marginTop:8}}>AI Cloud IDE</div>
                    <div style={{fontSize:12,marginTop:6,color:'var(--vscode-text-dim)'}}>Open a file from the Explorer or create a new one</div>
                    <div style={{fontSize:11,marginTop:16,color:'#555'}}>Ctrl+Shift+P to open the command palette</div>
                  </div>
                )}
              </Panel>
              <PanelResizeHandle className="resize-handle horizontal"/>
              <Panel defaultSize={30} minSize={15}>
                <div className="bottom-tabs">
                  <span className={`bottom-tab ${bottomTab==='terminal'?'active':''}`} onClick={()=>setBottomTab('terminal')}>TERMINAL</span>
                  <span className={`bottom-tab ${bottomTab==='problems'?'active':''}`} onClick={()=>setBottomTab('problems')}>
                    PROBLEMS {(errorCount>0||warnCount>0) && <span className="problem-badge" style={{background:errorCount>0?'#f44336':'#e3b341'}}>{errorCount+warnCount}</span>}
                  </span>
                  <span className={`bottom-tab ${bottomTab==='output'?'active':''}`} onClick={()=>setBottomTab('output')}>OUTPUT</span>
                </div>
                <div ref={terminalRef} className="terminal-host" style={{display:bottomTab==='terminal'?'block':'none'}}/>
                {bottomTab==='problems' && (
                  <div className="problems-host">
                    {!problems.length && <div style={{color:'var(--vscode-text-dim)'}}>✓ No problems detected.</div>}
                    {problems.map((p,i) => (
                      <div key={i} className={`problem-row ${p.severity}`}>
                        <span className="problem-icon">{p.severity==='error'?'✖':'⚠'}</span>
                        <span className="problem-file">{p.file}:{p.line}:{p.col}</span>
                        <span>{p.message}</span>
                      </div>
                    ))}
                  </div>
                )}
                {bottomTab==='output' && <div className="problems-host">{runOutput||'No output yet. Click ▶ Run.'}</div>}
              </Panel>
            </PanelGroup>
          </Panel>
        </PanelGroup>
      </div>

      <footer className="status-bar">
        <span>⎇ {gitBranch}</span>
        <span className={`docker-dot ${dockerStatus}`}>🐳 {dockerStatus}</span>
        <span className="spacer"/>
        {errorCount>0 && <span style={{color:'#f48771'}}>✖ {errorCount} error{errorCount>1?'s':''}</span>}
        {warnCount>0 && <span style={{color:'#e3b341'}}>⚠ {warnCount} warning{warnCount>1?'s':''}</span>}
        {errorCount===0&&warnCount===0 && <span style={{color:'#4caf50'}}>✓ No errors</span>}
        <span>{activeLang}</span>
        <span>UTF-8</span>
        <span>{chatLoading?'AI: thinking…':'AI: ready'}</span>
      </footer>

      {paletteOpen && (
        <div className="palette-overlay" onClick={()=>setPaletteOpen(false)}>
          <div className="palette-box" onClick={(e)=>e.stopPropagation()}>
            <input autoFocus className="palette-input" placeholder="Type a command or filename..."
              value={paletteQuery}
              onChange={(e)=>{setPaletteQuery(e.target.value);setPaletteIndex(0);}}
              onKeyDown={handlePaletteKeyDown}/>
            <div className="palette-list">
              {filteredCommands.map((cmd,i)=>(
                <div key={cmd.id} className={`palette-item ${i===paletteIndex?'active':''}`}
                  onClick={()=>runCommand(cmd)} onMouseEnter={()=>setPaletteIndex(i)}>
                  {cmd.label}
                </div>
              ))}
              {!filteredCommands.length && <div className="palette-empty">No matching commands.</div>}
            </div>
          </div>
        </div>
      )}
    </div>
  );
}

import { Navigate } from 'react-router-dom';
import { getToken } from './auth';

export default function RequireAuth({ children }) {
  const token = getToken();
  if (!token) return <Navigate to="/login" replace />;
  return children;
}
